```js
/** 累计结余：切换查看某个月份的结余（复用 balanceHistory 的单月结余）。 */
function shiftCumMonth(delta) {
  const balances = Store.balanceHistory(12);
  const currentKey = U.ymd(new Date()).slice(0, 7);
  let idx = -1;
  const key = state.cumMonth || currentKey;
  balances.forEach(function (b, i) { if (b.key === key) idx = i; });
  if (idx < 0) idx = balances.length - 1;
  const next = Math.min(balances.length - 1, Math.max(0, idx + delta));
  state.cumMonth = balances[next].key;
  const y = window.scrollY;
  render();
  window.scrollTo(0, y);
}

function pickCumMonth(key) {
  if (!key) return;
  state.cumMonth = key;
  const y = window.scrollY;
  render();
  window.scrollTo(0, y);
}
```

```js
/* 输入起止日期后立即刷新结果；不选时段时 from/to 为空 = 全局 */
document.addEventListener('change', function (event) {
  const target = event.target;
  if (!target || (target.id !== 'note-from' && target.id !== 'note-to')) return;
  const from = document.getElementById('note-from');
  const to = document.getElementById('note-to');
  state.noteFrom = from ? from.value : '';
  state.noteTo = to ? to.value : '';
  const clearBtn = document.querySelector('[data-action="note-range-clear"]');
```

## 三、自测步骤

### ① 洗衣分类

1. 首页点「记一笔」→ 面板出现 12 个分类，第 4 行左起第 1 个是「洗衣」（蓝色衬衫图标）。
2. 选「洗衣」→ 输入 6 → 保存；再记一笔 4 元。
3. 统计页 → 分类支出占比：出现“洗衣 ¥10.00”；月总支出 +10。
4. 我的 → 预算管理：出现“洗衣”预算行，可单独设置额度并保存。
5. 日历选中当天：账单列表显示两笔洗衣记录，删除/编辑正常；旧账单金额不变。

### ② 累计结余按月切换

1. 统计页 → 滚到「累计结余」。
2. 默认显示当前月（2026年10月）；点 ‹ → 标题变 2026年9月，金额变为 9 月结余；点 › 回到 10 月。
3. 点柱状图任意一根（如 11月）→ 标题与金额同步切到该月，被选中的柱子高亮。
4. 到最早/最晚月份时对应按钮变灰不可点；切换后页面滚动位置不跳回顶部。
5. 金额与“近 12 个月”柱高一致；未设预算的月份显示 ¥0.00。

### ③ 备注搜索时段

1. 统计页 → 备注搜索输入“奶茶”（不选日期）→ 显示全局匹配笔数与总花费。
2. 选“从 2026-10-15、到 2026-10-31” → 结果只统计该区间，文案出现“统计时段”，笔数与总花费同步变小。
3. 点“清除” → 日期清空，恢复全局结果，搜索词保留。
4. 只选开始日期（不选结束）→ 统计该日期之后的全部匹配账单；只选结束日期 → 统计该日期之前的。
5. 清空关键词 → 结果区隐藏，不报错。
